# Cartella delle immagini caricate (uploads/): qui non si esegue niente, mai.
# Lo scrive il pulsante "Scrivi uploads/.htaccess" della pagina Sicurezza del pannello; su Nginx vale la regola in INSTALLA.md.
Options -Indexes -ExecCGI
<IfModule mod_php.c>
  php_flag engine off
</IfModule>
<IfModule mod_php7.c>
  php_flag engine off
</IfModule>
<IfModule mod_php8.c>
  php_flag engine off
</IfModule>
RemoveHandler .php .phtml .php3 .php4 .php5 .php7 .php8 .phps .phar .pl .py .cgi .shtml
RemoveType .php .phtml .php3 .php4 .php5 .php7 .php8 .phps .phar
<FilesMatch "\.(php|phtml|php[0-9]|phps|phar|pl|py|cgi|sh|shtml|htaccess)$">
  <IfModule mod_authz_core.c>
    Require all denied
  </IfModule>
  <IfModule !mod_authz_core.c>
    Order allow,deny
    Deny from all
  </IfModule>
</FilesMatch>
# Gli SVG si aprono in una sandbox: anche se uno contenesse uno script, non girerebbe sul dominio del sito.
<IfModule mod_headers.c>
  <FilesMatch "\.svg$">
    Header set Content-Security-Policy "sandbox; script-src 'none'; default-src 'self' data:"
  </FilesMatch>
  Header set X-Content-Type-Options "nosniff"
</IfModule>
